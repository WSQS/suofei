    iptables -N SOPHO_CNT
    iptables -A SOPHO_CNT -p tcp --dport 443 --syn -m owner --uid-owner 10217 -j RETURN   # 只数 SYN
    iptables -A SOPHO_CNT -p tcp --dport 443 -m owner --uid-owner 10217 -j RETURN         # 数其余出向包
    iptables -I OUTPUT 1 -j SOPHO_CNT        # 每个包进链后立刻 RETURN 回 OUTPUT 原位置 ⇒ 原有规则顺序不变

idletimer label / routectrl mark / nflog prefix 这三处动态值），OUTPUT 链恢复为原 4 条：
oem_out / fw_OUTPUT / st_OUTPUT / bw_OUTPUT。
